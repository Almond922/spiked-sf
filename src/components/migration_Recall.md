#### 1. Start Meeting Recording
**POST** `/start`
- **Purpose**: Initiates recording of a meeting URL via Recall.ai
- **Content-Type**: `application/x-www-form-urlencoded`
- **Parameters**:
  ```
  meeting_url: string (required)
  ```
- **Example Request**:
  ```javascript
  const formData = new FormData();
  formData.append('meeting_url', 'https://meet.google.com/abc-def-ghi');
  
  fetch('/start', {
    method: 'POST',
    body: formData
  });
  ```
- **Response**: `{"status": "started"}`

#### 2. Real-time Transcripts Stream
**GET** `/transcripts`
- **Purpose**: Server-Sent Events (SSE) stream of live meeting transcripts
- **Response Format**: 
  ```javascript
  {
    "speaker": "John Doe",
    "text": "I'm interested in AWS Lambda pricing"
  }
  ```
- **Usage**:
  ```javascript
  const transcriptSource = new EventSource('/transcripts');
  transcriptSource.onmessage = (event) => {
    const data = JSON.parse(event.data);
    console.log(`${data.speaker}: ${data.text}`);
  };
  ```

#### 3. Real-time Questions Stream
**GET** `/questions`
- **Purpose**: SSE stream of AI-detected AWS-related questions
- **Response Format**:
  ```javascript
  {
    "speaker": "John Doe", 
    "question": "What is the pricing for AWS Lambda?<br>Explanation: Client is explicitly asking about Lambda pricing"
  }
  ```
- **Usage**:
  ```javascript
  const questionSource = new EventSource('/questions');
  questionSource.onmessage = (event) => {
    const data = JSON.parse(event.data);
    // Handle detected question
  };
  ```

#### 4. Webhook Endpoint (Internal)
**POST** `/webhook/recall/transcript`
- **Purpose**: Receives transcript data from Recall.ai (not for frontend use)
- **Note**: This endpoint processes incoming transcripts and triggers question detection.......

# React integration example for reference(generated by claude)

### Complete React Component
```jsx
import React, { useState, useEffect, useRef } from 'react';

const MeetingRecorder = () => {
  const [meetingUrl, setMeetingUrl] = useState('');
  const [transcripts, setTranscripts] = useState([]);
  const [questions, setQuestions] = useState([]);
  const [isRecording, setIsRecording] = useState(false);
  const [error, setError] = useState('');
  
  const transcriptSourceRef = useRef(null);
  const questionSourceRef = useRef(null);

  const startRecording = async (e) => {
    e.preventDefault();
    setError('');
    
    try {
      const formData = new FormData();
      formData.append('meeting_url', meetingUrl);
      
      const response = await fetch('/start', {
        method: 'POST',
        body: formData
      });
      
      if (!response.ok) {
        throw new Error('Failed to start recording');
      }
      
      setIsRecording(true);
      
      // Start listening to transcript stream
      transcriptSourceRef.current = new EventSource('/transcripts');
      transcriptSourceRef.current.onmessage = (event) => {
        const data = JSON.parse(event.data);
        setTranscripts(prev => [...prev, data]);
      };
      
      transcriptSourceRef.current.onerror = (error) => {
        console.error('Transcript stream error:', error);
        setError('Connection to transcript stream failed');
      };
      
      // Start listening to questions stream
      questionSourceRef.current = new EventSource('/questions');
      questionSourceRef.current.onmessage = (event) => {
        const data = JSON.parse(event.data);
        setQuestions(prev => [...prev, data]);
      };
      
      questionSourceRef.current.onerror = (error) => {
        console.error('Question stream error:', error);
        setError('Connection to questions stream failed');
      };
      
    } catch (err) {
      setError(err.message);
    }
  };

  const stopRecording = () => {
    if (transcriptSourceRef.current) {
      transcriptSourceRef.current.close();
    }
    if (questionSourceRef.current) {
      questionSourceRef.current.close();
    }
    setIsRecording(false);
  };

  useEffect(() => {
    // Cleanup on component unmount
    return () => {
      if (transcriptSourceRef.current) {
        transcriptSourceRef.current.close();
      }
      if (questionSourceRef.current) {
        questionSourceRef.current.close();
      }
    };
  }, []);

  return (
    <div className="meeting-recorder">
      <div className="controls">
        <form onSubmit={startRecording}>
          <input
            type="text"
            value={meetingUrl}
            onChange={(e) => setMeetingUrl(e.target.value)}
            placeholder="https://meet.google.com/..."
            disabled={isRecording}
            required
          />
          <button type="submit" disabled={isRecording || !meetingUrl}>
            {isRecording ? 'Recording...' : 'Start Recording'}
          </button>
          {isRecording && (
            <button type="button" onClick={stopRecording}>
              Stop Recording
            </button>
          )}
        </form>
        
        {error && <div className="error">{error}</div>}
      </div>

      <div className="content">
        <div className="transcripts-panel">
          <h3>Live Transcript</h3>
          <div className="messages">
            {transcripts.map((item, index) => (
              <div key={index} className="message">
                <strong>{item.speaker}:</strong> {item.text}
              </div>
            ))}
          </div>
        </div>

        <div className="questions-panel">
          <h3>Detected Questions</h3>
          <div className="messages">
            {questions.map((item, index) => (
              <div key={index} className="question">
                <strong>{item.speaker}:</strong>
                <div dangerouslySetInnerHTML={{ __html: item.question }} />
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
};

export default MeetingRecorder;
```
